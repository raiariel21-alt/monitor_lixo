O Rio São Francisco

🌊 O Velho Chico

O Rio São Francisco é um dos rios mais importantes do Brasil e da América do Sul. Chamado carinhosamente de Velho Chico, é a principal fonte de água potável para milhões de pessoas no semiárido nordestino, onde a escassez hídrica é crítica.

Nasce na Serra da Canastra, em Minas Gerais, a 1.550 m de altitude, e percorre 2.863 km até desaguar no Oceano Atlântico, na divisa entre Alagoas e Sergipe. É considerado o "rio da unidade nacional" por ser o único grande rio inteiramente brasileiro que liga o Centro-Sul ao Nordeste.

Apesar de sua importância histórica e ecológica, o rio sofre com décadas de degradação ambiental: desmatamento das matas ciliares, lançamento de esgoto sem tratamento, mineração irregular e descarte de lixo nas margens ameaçam sua sobrevivência.

Problemas de poluição
Principais ameaças ao rio
Descarte irregular de lixo
Resíduos sólidos domésticos, plásticos e entulhos são descartados diretamente nas margens do rio por comunidades sem coleta de lixo adequada.
Esgoto sem tratamento
Mais de 80% das cidades ribeirinhas lançam esgoto doméstico e industrial diretamente no rio, contaminando a água com patógenos e metais pesados.
Agrotóxicos e fertilizantes
O uso intensivo na agricultura do entorno contamina os afluentes com nitratos, fosfatos e pesticidas que chegam ao rio principal.
Desmatamento das margens
A retirada da mata ciliar provoca erosão, assoreamento do leito e aumenta drasticamente a turbidez da água, afetando a vida aquática.
Ações de recuperação em curso
Programas de reflorestamento, tratamento de esgoto e fiscalização ambiental estão sendo ampliados ao longo do rio — mas ainda são insuficientes.

Sobre o projeto
Microcontrolador ESP32
Chip com Wi-Fi integrado que coleta os dados do sensor e envia via HTTP ao servidor a cada 5 segundos. Alimentado por bateria ou USB.
Sensor Ultrassônico HC-SR04
Mede distância com precisão de 3 mm. Objetos detectados a menos de 100 cm são classificados como presença de lixo na área monitorada.
Alertas em tempo real
Quando lixo é detectado, o painel exibe alerta imediato com horário e distância. Futuramente integrado com SMS e e-mail para equipes de campo.
Mapeamento georreferenciado
Cada sensor tem coordenadas GPS. O mapa interativo mostra em tempo real os pontos com lixo (vermelho) ou livres (verde) ao longo do rio.
Histórico e análise de dados
Todas as leituras são armazenadas com horário e distância. Gráficos mostram padrões de descarte por hora, dia e semana para planejamento de ações.
Impacto ambiental esperado
Reduzir o acúmulo de resíduos nas margens do Rio São Francisco, proteger a biodiversidade aquática e garantir a qualidade da água para comunidades ribeirinhas.

Código completo do ESP32 + servidor
// ══════════════════════════════════════════════════
//  Monitor de Lixo — Rio São Francisco
//  ESP32 + Sensor Ultrassônico HC-SR04
//  Envia dados via Wi-Fi para o servidor
// ══════════════════════════════════════════════════

#include <WiFi.h>
#include <HTTPClient.h>

// ── Configurações Wi-Fi ──────────────────────────
const char* SSID      = "SUA_REDE_WIFI";
const char* PASSWORD  = "SUA_SENHA";
const char* SERVER    = "http://seusite.com/api/sensor";

// ── Pinos do HC-SR04 ─────────────────────────────
#define TRIG_PIN    5    // GPIO 5
#define ECHO_PIN   18    // GPIO 18
#define LED_PIN     2    // LED interno ESP32

// ── Configurações do sensor ──────────────────────
#define LIMIAR_CM  100.0  // abaixo = lixo detectado
#define SENSOR_ID  "A1"
#define LATITUDE   -10.5014   // coordenadas do ponto
#define LONGITUDE  -36.9614

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);

  Serial.println("Conectando ao Wi-Fi...");
  WiFi.begin(SSID, PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500); Serial.print(".");
  }
  Serial.println("\nWi-Fi conectado!");
  Serial.print("IP: "); Serial.println(WiFi.localIP());
  digitalWrite(LED_PIN, HIGH); // LED aceso = online
}

float medir() {
  digitalWrite(TRIG_PIN, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long dur = pulseIn(ECHO_PIN, HIGH, 30000);
  return dur == 0 ? 999.0 : dur * 0.034 / 2.0;
}

void loop() {
  float dist    = medir();
  bool  temLixo = dist < LIMIAR_CM;

  // Pisca 3x se tiver lixo
  if (temLixo) {
    for (int i = 0; i < 3; i++) {
      digitalWrite(LED_PIN, LOW);  delay(150);
      digitalWrite(LED_PIN, HIGH); delay(150);
    }
  }

  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(SERVER);
    http.addHeader("Content-Type", "application/json");

    String body = "{"
      "\"sensor\":\""   + String(SENSOR_ID)    + "\","
      "\"distancia\":"  + String(dist, 1)       + ","
      "\"lixo\":"       + (temLixo ? "true" : "false") + ","
      "\"latitude\":"   + String(LATITUDE,  4)  + ","
      "\"longitude\":"  + String(LONGITUDE, 4)  +
    "}";

    int code = http.POST(body);
    Serial.printf("%s %.1f cm → HTTP %d\n",
      temLixo ? "[LIXO]" : "[OK]  ", dist, code);
    http.end();
  }

  delay(5000); // envia a cada 5 segundos
}
# ══════════════════════════════════════════════
#  API Backend — FastAPI + SQLite
#  Recebe dados do ESP32 e serve o frontend
# ══════════════════════════════════════════════

from fastapi import FastAPI
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel
from datetime import datetime
import sqlite3

app = FastAPI(title="Monitor Rio São Francisco")
app.mount("/static", StaticFiles(directory="."), name="static")

class Leitura(BaseModel):
    sensor: str; distancia: float; lixo: bool
    latitude:  float = -10.5014
    longitude: float = -36.9614

def db():
    c = sqlite3.connect("monitor.db")
    c.row_factory = sqlite3.Row
    return c

# Cria tabela ao iniciar
with db() as c:
    c.execute("""
        CREATE TABLE IF NOT EXISTS leituras (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            sensor_id TEXT, distancia REAL, lixo INTEGER,
            latitude REAL, longitude REAL, criado_em TEXT)
    """)

@app.post("/api/sensor")
async def receber(d: Leitura):
    with db() as c:
        c.execute("INSERT INTO leituras VALUES(NULL,?,?,?,?,?,?)",
            (d.sensor, d.distancia, int(d.lixo),
             d.latitude, d.longitude, datetime.now().isoformat()))
    return {"ok": True}

@app.get("/api/status")
async def status():
    with db() as c:
        rows = c.execute("""SELECT * FROM leituras WHERE id IN
            (SELECT MAX(id) FROM leituras GROUP BY sensor_id)""").fetchall()
    return [dict(r) for r in rows]

@app.get("/api/historico")
async def historico(limite: int = 50):
    with db() as c:
        rows = c.execute(
            "SELECT * FROM leituras ORDER BY id DESC LIMIT ?",
            (limite,)).fetchall()
    return [dict(r) for r in rows]

@app.get("/api/estatisticas")
async def estatisticas():
    with db() as c:
        total     = c.execute("SELECT COUNT(*) FROM leituras").fetchone()[0]
        com_lixo  = c.execute("SELECT COUNT(*) FROM leituras WHERE lixo=1").fetchone()[0]
        por_hora  = c.execute("""
            SELECT strftime('%H', criado_em) hora,
                   COUNT(*) total, SUM(lixo) com_lixo
            FROM leituras WHERE date(criado_em)=date('now')
            GROUP BY hora ORDER BY hora""").fetchall()
    taxa = round(com_lixo/total*100, 1) if total else 0
    return {"total": total, "total_lixo": com_lixo,
            "taxa_ocupacao": taxa, "por_hora": [dict(r) for r in por_hora]}

@app.get("/")
async def index(): return FileResponse("index.html")

-- Banco de dados SQLite — Monitor Rio São Francisco
-- Criado automaticamente pelo servidor Python
-- ═══════════════════════════════════════════════

CREATE TABLE IF NOT EXISTS leituras (
    id         INTEGER PRIMARY KEY AUTOINCREMENT,
    sensor_id  TEXT    NOT NULL,          -- ex: "A1"
    distancia  REAL    NOT NULL,          -- em cm
    lixo       INTEGER NOT NULL,          -- 0=livre, 1=lixo
    latitude   REAL    DEFAULT -10.5014, -- coordenada GPS
    longitude  REAL    DEFAULT -36.9614,
    criado_em  TEXT    NOT NULL           -- ISO 8601
);

-- Status atual de cada sensor (última leitura)
SELECT * FROM leituras
WHERE id IN (
    SELECT MAX(id) FROM leituras GROUP BY sensor_id
);

-- Detecções de hoje agrupadas por hora
SELECT strftime('%H', criado_em) AS hora,
       COUNT(*) AS total,
       SUM(lixo) AS com_lixo
FROM leituras
WHERE date(criado_em) = date('now')
GROUP BY hora ORDER BY hora;

-- Taxa de ocupação geral (%)
SELECT ROUND(SUM(lixo) * 100.0 / COUNT(*), 1) AS taxa_pct
FROM leituras;

-- Alertas das últimas 24 horas
SELECT * FROM leituras
WHERE lixo = 1
  AND criado_em >= datetime('now', '-24 hours')
ORDER BY id DESC;
